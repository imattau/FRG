> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Implement the FRG Core library — serialisation, hashing, tree construction, and error handling — fully compliant with the Fractal RG Ledger Protocol v1.0.0 spec.

**Architecture:** Layered sub-packages (`errors` → `hash` → `tx`/`node` → `tree`) with unidirectional dependencies. Pure functions as the foundation, with a `Block` convenience wrapper in `tree`. Variance computed via sum propagation (sumValues, sumSquares, count) during bottom-up tree construction.

**Tech Stack:** Go 1.22+, `math/big` (uint256), `golang.org/x/text/unicode/norm` (NFC normalisation), `pgregory.net/rapid` (property tests), standard `crypto/sha256`.

package hash

import (
	"crypto/sha256"
	"math/big"
)

const (
	DomainTx         = "\x54\x58\x5F\x56\x31\x00"
	DomainRGNode     = "\x52\x47\x5F\x4E\x4F\x44\x45\x5F\x56\x31\x00"
	DomainNullPad    = "\x4E\x55\x4C\x4C\x5F\x50\x41\x44\x5F\x56\x31\x00"
	DomainEmptyBlock = "\x45\x4D\x50\x54\x59\x5F\x42\x4C\x4F\x43\x4B\x5F\x56\x31\x00"
)

// UINT256_MAX = 2^256 - 1
var UINT256_MAX = new(big.Int).Sub(new(big.Int).Lsh(big.NewInt(1), 256), big.NewInt(1))

// validScales contains all valid Λ = 4^n for n ∈ {0..8}
var validScales = map[uint32]bool{
	1: true, 4: true, 16: true, 64: true, 256: true,
	1024: true, 4096: true, 16384: true, 65536: true,
}

// Hash performs a single-pass SHA2-256 over data.
func Hash(data []byte) [32]byte {
	return sha256.Sum256(data)
}

// ValidScale returns true if lambda = K^n for n ∈ {0..8}.
func ValidScale(lambda uint32) bool {
	return validScales[lambda]
}
```

package tx

import (
	"encoding/binary"
	"math/big"

	rgerrors "github.com/imattau/frg/core/errors"
	"github.com/imattau/frg/core/hash"
	"golang.org/x/text/unicode/norm"
)

const maxTxBytes = 70000

// Tx represents a single transaction.
// Value must be pre-scaled by 10^18 before construction.
type Tx struct {
	Sender   string
	Receiver string
	Value    *big.Int // uint256, pre-scaled by SCALE = 10^18
	Nonce    uint64
}

// Serialize returns Tx_Bytes per spec:
//
//	TX_V1\x00 | uint16(len(Sender)) | Sender | uint16(len(Receiver)) | Receiver | uint256(Value) | uint64(Nonce)
func (t *Tx) Serialize() ([]byte, error) {
	sender := norm.NFC.String(t.Sender)
	receiver := norm.NFC.String(t.Receiver)

	if len(sender) > 65535 || len(receiver) > 65535 {
		return nil, rgerrors.New(rgerrors.ErrCanonicalEncodingDistortion, "sender or receiver exceeds uint16 length")
	}

	size := 6 + 2 + len(sender) + 2 + len(receiver) + 32 + 8
	if size > maxTxBytes {
		return nil, rgerrors.Newf(rgerrors.ErrDosSizeExceeded, "tx payload %d bytes exceeds 70000", size)
	}

	if t.Value == nil || t.Value.Sign() < 0 || t.Value.Cmp(hash.UINT256_MAX) > 0 {
		return nil, rgerrors.New(rgerrors.ErrArithmeticOverflow, "value out of uint256 range")
	}

	buf := make([]byte, size)
	pos := 0

	pos += copy(buf[pos:], hash.DomainTx)

	binary.BigEndian.PutUint16(buf[pos:], uint16(len(sender)))
	pos += 2
	pos += copy(buf[pos:], sender)

	binary.BigEndian.PutUint16(buf[pos:], uint16(len(receiver)))
	pos += 2
	pos += copy(buf[pos:], receiver)

	// uint256 big-endian, 32 bytes
	vBytes := t.Value.Bytes()
	if len(vBytes) > 32 {
		return nil, rgerrors.New(rgerrors.ErrArithmeticOverflow, "value exceeds uint256")
	}
	pos += 32 - len(vBytes) // zero-pad left
	copy(buf[pos:], vBytes)
	pos += len(vBytes)

	binary.BigEndian.PutUint64(buf[pos:], t.Nonce)

	return buf, nil
}

// ID returns txid = H(Tx_Bytes).
func (t *Tx) ID() ([32]byte, error) {
	b, err := t.Serialize()
	if err != nil {
		return [32]byte{}, err
	}
	return hash.Hash(b), nil
}
```

package node

import (
	"encoding/binary"
	"math/big"

	rgerrors "github.com/imattau/frg/core/errors"
	"github.com/imattau/frg/core/hash"
)

type Signature uint8

const (
	SigAtomic        Signature = 1
	SigNullPad       Signature = 2
	SigStagnantState Signature = 3
	SigLaminarFlow   Signature = 4
	SigVolatileShock Signature = 5
)

// RGNode represents a node in the RG tree.
// sumValues, sumSquares, count are internal propagation fields — not serialised.
type RGNode struct {
	Scale    uint32
	Volume   *big.Int
	Variance *big.Int
	Sig      Signature
	Children [][32]byte

	SumValues  *big.Int
	SumSquares *big.Int
	Count      uint64
}

// Serialize returns Node_Bytes per spec.
func (n *RGNode) Serialize() ([]byte, error) {
	if !hash.ValidScale(n.Scale) {
		return nil, rgerrors.Newf(rgerrors.ErrScaleDomainFault, "invalid scale %d", n.Scale)
	}

	expectedChildren := 4
	if n.Scale == 1 {
		expectedChildren = 1
	}
	if len(n.Children) != expectedChildren {
		return nil, rgerrors.Newf(rgerrors.ErrInvalidChildArity, "expected %d children, got %d", expectedChildren, len(n.Children))
	}

	if n.Volume == nil || n.Volume.Sign() < 0 || n.Volume.Cmp(hash.UINT256_MAX) > 0 {
		return nil, rgerrors.New(rgerrors.ErrArithmeticOverflow, "volume out of uint256 range")
	}
	if n.Variance == nil || n.Variance.Sign() < 0 || n.Variance.Cmp(hash.UINT256_MAX) > 0 {
		return nil, rgerrors.New(rgerrors.ErrArithmeticOverflow, "variance out of uint256 range")
	}

	// RG_NODE_V1\x00(11) + scale(4) + volume(32) + variance(32) + sig(1) + child_count(2) + children(M*32)
	size := 11 + 4 + 32 + 32 + 1 + 2 + len(n.Children)*32
	buf := make([]byte, size)
	pos := 0

	pos += copy(buf[pos:], hash.DomainRGNode)
	binary.BigEndian.PutUint32(buf[pos:], n.Scale)
	pos += 4

	vb := n.Volume.Bytes()
	pos += 32 - len(vb)
	copy(buf[pos:], vb)
	pos += len(vb)

	varb := n.Variance.Bytes()
	pos += 32 - len(varb)
	copy(buf[pos:], varb)
	pos += len(varb)

	buf[pos] = byte(n.Sig)
	pos++

	binary.BigEndian.PutUint16(buf[pos:], uint16(len(n.Children)))
	pos += 2

	for _, c := range n.Children {
		copy(buf[pos:], c[:])
		pos += 32
	}

	return buf, nil
}

// Root returns node_root = H(Node_Bytes).
func (n *RGNode) Root() ([32]byte, error) {
	b, err := n.Serialize()
	if err != nil {
		return [32]byte{}, err
	}
	return hash.Hash(b), nil
}

// RecomputeSig derives the correct Signature from Volume and Variance.
// Rules:
//
//	volume == 0 && variance == 0 → SigStagnantState
//	variance == 0               → SigLaminarFlow
//	variance > volume*volume    → SigVolatileShock  (high relative variance)
//	else                        → SigLaminarFlow
//
// For atomic nodes (Scale==1): always SigAtomic.
// For null-pad nodes: always SigNullPad.
// Returns ERR_007 if recomputed sig != n.Sig.
func (n *RGNode) RecomputeSig() (Signature, error) {
	computed := computeSig(n)
	if computed != n.Sig {
		return 0, rgerrors.Newf(rgerrors.ErrSignatureMisrepresentation,
			"declared sig %d, recomputed %d", n.Sig, computed)
	}
	return computed, nil
}

func computeSig(n *RGNode) Signature {
	if n.Scale == 1 {
		return SigAtomic
	}
	zero := big.NewInt(0)
	if n.Volume.Cmp(zero) == 0 && n.Variance.Cmp(zero) == 0 {
		return SigStagnantState
	}
	if n.Variance.Cmp(zero) == 0 {
		return SigLaminarFlow
	}
	// VolatileShock: variance > volume (heuristic — variance as fraction of volume)
	if n.Variance.Cmp(n.Volume) > 0 {
		return SigVolatileShock
	}
	return SigLaminarFlow
}

// NullNode returns the canonical NULL_Λ node at the given scale.
func NullNode(scale uint32) (*RGNode, error) {
	if !hash.ValidScale(scale) {
		return nil, rgerrors.Newf(rgerrors.ErrScaleDomainFault, "invalid scale %d", scale)
	}

	// child = H(NULL_PAD_V1\x00)
	childRoot := hash.Hash([]byte(hash.DomainNullPad))

	childCount := 1
	if scale > 1 {
		childCount = 4
	}
	children := make([][32]byte, childCount)
	for i := range children {
		children[i] = childRoot
	}

	return &RGNode{
		Scale:      scale,
		Volume:     big.NewInt(0),
		Variance:   big.NewInt(0),
		Sig:        SigNullPad,
		Children:   children,
		SumValues:  big.NewInt(0),
		SumSquares: big.NewInt(0),
		Count:      0,
	}, nil
}

// EmptyBlockRoot returns the normative empty-block state root constant.
func EmptyBlockRoot() [32]byte {
	childRoot := hash.Hash([]byte(hash.DomainEmptyBlock))
	n := &RGNode{
		Scale:    1,
		Volume:   big.NewInt(0),
		Variance: big.NewInt(0),
		Sig:      SigStagnantState,
		Children: [][32]byte{childRoot},
	}
	root, _ := n.Root()
	return root
}

package tree

import (
	"math/big"

	rgerrors "github.com/imattau/frg/core/errors"
	"github.com/imattau/frg/core/node"
	"github.com/imattau/frg/core/tx"
)

const (
	K      = 4
	TMax   = 65536
)

// Block represents a single block of transactions.
// Height is an out-of-band routing parameter — not serialised, used for ERR_008.
type Block struct {
	Height uint64
	Txs    []*tx.Tx
}

// BuildRoot executes the full bottom-up pipeline and returns the state root.
func (b *Block) BuildRoot() ([32]byte, error) {
	if len(b.Txs) == 0 {
		return node.EmptyBlockRoot(), nil
	}
	if len(b.Txs) > TMax {
		return [32]byte{}, rgerrors.Newf(rgerrors.ErrDosSizeExceeded, "block has %d txs, max %d", len(b.Txs), TMax)
	}

	// Step 1: Atomic ingestion
	layer, err := atomicLayer(b.Txs)
	if err != nil {
		return [32]byte{}, err
	}

	// Step 2+3: Chunk and coarse-grain until single root
	for len(layer) > 1 {
		layer, err = coarsenLayer(layer)
		if err != nil {
			return [32]byte{}, err
		}
	}

	return layer[0].Root()
}

func atomicLayer(txs []*tx.Tx) ([]*node.RGNode, error) {
	nodes := make([]*node.RGNode, len(txs))
	for i, t := range txs {
		id, err := t.ID()
		if err != nil {
			return nil, err
		}
		vol := new(big.Int).Set(t.Value)
		nodes[i] = &node.RGNode{
			Scale:      1,
			Volume:     vol,
			Variance:   big.NewInt(0),
			Sig:        node.SigAtomic,
			Children:   [][32]byte{id},
			SumValues:  new(big.Int).Set(t.Value),
			SumSquares: new(big.Int).Mul(t.Value, t.Value),
			Count:      1,
		}
	}
	return nodes, nil
}

func coarsenLayer(layer []*node.RGNode) ([]*node.RGNode, error) {
	parentScale := layer[0].Scale * K
	var parents []*node.RGNode

	for i := 0; i < len(layer); i += K {
		end := i + K
		needPad := 0
		if end > len(layer) {
			needPad = end - len(layer)
			end = len(layer)
		}
		chunk := make([]*node.RGNode, end-i, K)
		copy(chunk, layer[i:end])

		// padding_mask: bit i set for each appended null
		paddingMask := uint8(0)
		for p := 0; p < needPad; p++ {
			pos := (K - needPad) + p
			paddingMask |= 1 << pos
			nullN, err := node.NullNode(layer[0].Scale)
			if err != nil {
				return nil, err
			}
			chunk = append(chunk, nullN)
		}

		if paddingMask >= (1 << K) {
			return nil, rgerrors.Newf(rgerrors.ErrMaskOutOfBounds, "padding_mask %d >= %d", paddingMask, 1<<K)
		}

		// Validate padding positions
		for p := 0; p < K; p++ {
			if (paddingMask>>p)&1 == 1 {
				expectedNull, err := node.NullNode(layer[0].Scale)
				if err != nil {
					return nil, err
				}
				expectedRoot, err := expectedNull.Root()
				if err != nil {
					return nil, err
				}
				actualRoot, err := chunk[p].Root()
				if err != nil {
					return nil, err
				}
				if actualRoot != expectedRoot {
					return nil, rgerrors.New(rgerrors.ErrPaddingSubstitutionFraud, "masked child is not canonical NULL_Λ")
				}
			}
		}

		parent, err := coarseGrain(chunk, parentScale)
		if err != nil {
			return nil, err
		}
		parents = append(parents, parent)
	}

	return parents, nil
}

func coarseGrain(chunk []*node.RGNode, parentScale uint32) (*node.RGNode, error) {
	sumValues := big.NewInt(0)
	sumSquares := big.NewInt(0)
	count := uint64(0)
	children := make([][32]byte, K)

	for i, child := range chunk {
		root, err := child.Root()
		if err != nil {
			return nil, err
		}
		children[i] = root

		sumValues.Add(sumValues, child.SumValues)
		sumSquares.Add(sumSquares, child.SumSquares)
		count += child.Count

		if sumValues.Cmp(big.NewInt(0)) < 0 || sumValues.BitLen() > 256 {
			return nil, rgerrors.New(rgerrors.ErrArithmeticOverflow, "sumValues overflow")
		}
		if sumSquares.Cmp(big.NewInt(0)) < 0 || sumSquares.BitLen() > 256 {
			return nil, rgerrors.New(rgerrors.ErrArithmeticOverflow, "sumSquares overflow")
		}
	}

	volume := new(big.Int).Set(sumValues)
	variance := big.NewInt(0)
	if count > 0 {
		n := new(big.Int).SetUint64(count)
		// variance = floor(sumSquares/n) - floor(sumValues/n)^2
		meanSq := new(big.Int).Div(sumSquares, n)
		mean := new(big.Int).Div(sumValues, n)
		meanSquared := new(big.Int).Mul(mean, mean)
		variance.Sub(meanSq, meanSquared)
		if variance.Sign() < 0 {
			return nil, rgerrors.New(rgerrors.ErrArithmeticOverflow, "variance is negative")
		}
	}

	parent := &node.RGNode{
		Scale:      parentScale,
		Volume:     volume,
		Variance:   variance,
		Children:   children,
		SumValues:  new(big.Int).Set(sumValues),
		SumSquares: new(big.Int).Set(sumSquares),
		Count:      count,
	}

	// Compute and assign signature
	sig, err := computeAndAssignSig(parent)
	if err != nil {
		return nil, err
	}
	parent.Sig = sig

	return parent, nil
}

func computeAndAssignSig(n *node.RGNode) (node.Signature, error) {
	zero := big.NewInt(0)
	if n.Volume.Cmp(zero) == 0 && n.Variance.Cmp(zero) == 0 {
		return node.SigStagnantState, nil
	}
	if n.Variance.Cmp(zero) == 0 {
		return node.SigLaminarFlow, nil
	}
	if n.Variance.Cmp(n.Volume) > 0 {
		return node.SigVolatileShock, nil
	}
	return node.SigLaminarFlow, nil
}
